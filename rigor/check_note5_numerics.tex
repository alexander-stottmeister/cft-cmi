\begin{abstract}\noindent
This document verifies, in Lamport style, (i) the formulas that the numerics in
\file{cft_cmi/numerics/} are supposed to implement and (ii) the code that implements them, and it
re-derives every number quoted in \S6 and in \file{fidelity_tables.tex}.  The headline results of
\S6 survive: an \emph{independent} root-fidelity evaluation reproduces
$\Phi(1/15)_{\rm raw}=3.432123\times10^{-5}$ and $\Phi(1/15)_{\rm cor}=3.519443\times10^{-5}$ to all
printed digits, the defect matrix is confirmed entrywise against a closed form obtained
independently in \file{rigor/kernel_identity.tex}, and the extrapolated $f_2$ agrees with the
analytic $1/(12\pi^2)$ to $1.5\times10^{-7}$.  Four defects are recorded: a genuine
\emph{basis inconsistency} between \file{compression_box.py} (centred box modes) and
\file{defect_matrix.py} (uncentred box modes), which voids the ``rigorous lower bound'' claim as
stated although it moves $\Phi$ only by $4.8\times10^{-5}$ relative; a false statement in \S3.1
about the spectrum of $Q_N$, contradicted by \S6 itself and by the code's own output; a
non-monotone convergence table presented as monotone; and a factor-$2$ error in \S7.2.  The tail
correction $\zeta^2/(15\kc^2)$ is derived and found to be right in structure but biased by
$10$--$50\%$ at $\eps=10^{-8}$, so that the ``$1.4\times10^{-4}$ agreement'' advertised as its
validation is a cancellation of two errors.
\end{abstract}

\section{The compressed vacuum symbol}\label{sec:qbox}

\begin{lemma}[Box compression of the thermal kernel]\label{lem:qbox}
Let $\widetilde q(\xi,\xi')=\tfrac12\delta(\xi-\xi')-\frac i{4\pi}\operatorname{PV}\frac1{\sinh\frac{\xi-\xi'}2}$
be the kernel \eqref{eq:conv} of $Q$ in the modular coordinate.  In the \emph{centred} basis
$\check e_j$ of \eqref{eq:bases},
\begin{align}
 \check Q_{jj}&=\frac12-\frac1{2\pi\Lambda}\int_0^\Lambda\frac{(\Lambda-t)\sin(k_jt)}{\sinh(t/2)}\,dt,
 \label{eq:qdiag}\\
 \check Q_{jl}&=-\frac{(-1)^{j-l}}{2\pi\Lambda\,(k_j-k_l)}\int_0^\Lambda
        \frac{\cos(k_jt)-\cos(k_lt)}{\sinh(t/2)}\,dt\qquad(j\neq l),\label{eq:qoff}
\end{align}
both integrands being regular at $t=0$, with $\lim_{t\to0}\frac{(\Lambda-t)\sin(k_jt)}{\sinh(t/2)}=2\Lambda k_j$.
In the \emph{uncentred} basis $e_j$ the same expressions carry the extra factor $e^{-i(k_j-k_l)\xi_0}$.
\end{lemma}

\lstep{1}{1}{\ldefine $T_{jl}:=\langle\check e_j,(Q-\tfrac12)\check e_l\rangle
 =-\frac i{4\pi\Lambda}\operatorname{PV}\!\!\iint_{[-\Lambda/2,\Lambda/2]^2}\!\!
 \frac{e^{-ik_ja+ik_lb}}{\sinh\frac{a-b}2}\,da\,db$, and
 $\langle\check e_j,\tfrac12\check e_l\rangle=\tfrac12\delta_{jl}$.}
\lby{\eqref{eq:bases}, the substitution $a=\xi-\xi_0$, $b=\xi'-\xi_0$ (which maps $B_\Lambda$ onto
 $[-\Lambda/2,\Lambda/2]$), and
 $\Lambda^{-1}\int_{-\Lambda/2}^{\Lambda/2}e^{i(k_l-k_j)a}da=\delta_{jl}$ because
 $k_l-k_j=2\pi(l-j)/\Lambda$}

\lstep{1}{2}{The Fourier symbol of $\widetilde q$ is $\widehat q(k)=1/(1+e^{2\pi k})=1/(1+e^{\kappa})$,
 $\kappa=2\pi k$; i.e.\ the normalisation of the $\tfrac12\delta$ and $-\frac i{4\pi}$ terms is the
 one required by \eqref{eq:fermi} of the note.}
\lproof
\lstep{2}{1}{$\int_0^\infty\frac{\sin(kt)}{\sinh(bt)}dt=\frac{\pi}{2b}\tanh\frac{\pi k}{2b}$; with $b=\tfrac12$,
 $\int_0^\infty\frac{\sin(kt)}{\sinh(t/2)}dt=\pi\tanh(\pi k)$.}
\lby{Gradshteyn--Ryzhik 3.981.1}
\lstep{2}{2}{$\operatorname{PV}\int_{\mathbb R}\frac{e^{-ikt}}{\sinh(t/2)}dt=-2i\pi\tanh(\pi k)$,
 because the $\cos$ part is an odd PV integral and vanishes.}
\lby{$\langle2\rangle1$}
\lstep{2}{3}{Hence the symbol is $\tfrac12-\frac i{4\pi}\bigl(-2i\pi\tanh(\pi k)\bigr)
 =\tfrac12\bigl(1-\tanh(\pi k)\bigr)=\frac1{1+e^{2\pi k}}$.}
\lby{$\langle2\rangle2$}
\lqed{2}{4}\lby{$\langle2\rangle3$}

\lstep{1}{3}{With $t=a-b$ and $b$ ranging over an interval of length $\Lambda-|t|$,
 $T_{jl}=-\frac i{4\pi\Lambda}\operatorname{PV}\!\int_{-\Lambda}^{\Lambda}\!dt\,
  \frac{e^{-ik_jt}}{\sinh(t/2)}\,\Xi_{jl}(t)$, $\ \Xi_{jl}(t)=\int_{b_-(t)}^{b_+(t)}e^{-i\Delta b}db$,
  $\Delta:=k_j-k_l$, $b_\pm$ as in $\langle1\rangle4$.}
\lby{Fubini for the (absolutely convergent, after the PV pairing) $t$-slicing of the square, and
 $e^{-ik_ja+ik_lb}=e^{-ik_jt}e^{-i\Delta b}$}

\lstep{1}{4}{$\Xi_{jj}(t)=\Lambda-|t|$, and for $j\neq l$,
 $\Xi_{jl}(t)=(-1)^{j-l}\,\operatorname{sgn}(t)\,\dfrac{1-e^{i\Delta t}}{i\Delta}$.}
\lproof
\lstep{2}{1}{For $t>0$: $b\in[-\Lambda/2,\Lambda/2-t]$, so
 $\Xi_{jl}=\bigl(e^{-i\Delta(\Lambda/2-t)}-e^{i\Delta\Lambda/2}\bigr)/(-i\Delta)$.}
\lby{elementary integration}
\lstep{2}{2}{$e^{\pm i\Delta\Lambda/2}=e^{\pm i\pi(j-l)}=(-1)^{j-l}$.}
\lby{$\Delta\Lambda/2=\pi(j-l)$}
\lstep{2}{3}{Hence for $t>0$, $\Xi_{jl}=(-1)^{j-l}(1-e^{i\Delta t})/(i\Delta)$; the case $t<0$ gives
 the same with $i\Delta\to-i\Delta$, i.e.\ the factor $\operatorname{sgn}(t)$.}
\lby{$\langle2\rangle1$, $\langle2\rangle2$ and the same computation on $b\in[-\Lambda/2-t,\Lambda/2]$}
\lqed{2}{4}\lby{$\langle2\rangle3$; the case $j=l$ is the length of the $b$-interval}

\lstep{1}{5}{$T_{jj}=-\frac1{2\pi\Lambda}\displaystyle\int_0^\Lambda\frac{(\Lambda-t)\sin(k_jt)}{\sinh(t/2)}dt$,
 which is \eqref{eq:qdiag}.}
\lproof
\lstep{2}{1}{$(\Lambda-|t|)\cos(k_jt)/\sinh(t/2)$ is odd, so its PV integral over $[-\Lambda,\Lambda]$
 vanishes; $-i(\Lambda-|t|)\sin(k_jt)/\sinh(t/2)$ is even.}
\lby{parity of $\sinh$, $\sin$, $\cos$}
\lstep{2}{2}{Hence $\operatorname{PV}\int_{-\Lambda}^\Lambda(\Lambda-|t|)e^{-ik_jt}/\sinh(t/2)dt
 =-2i\int_0^\Lambda(\Lambda-t)\sin(k_jt)/\sinh(t/2)dt$.}
\lby{$\langle2\rangle1$}
\lstep{2}{3}{$-\frac i{4\pi\Lambda}\cdot(-2i)=-\frac1{2\pi\Lambda}$.}
\lby{arithmetic}
\lqed{2}{4}\lby{$\langle1\rangle3$, $\langle2\rangle2$, $\langle2\rangle3$}

\lstep{1}{6}{$T_{jl}=-\dfrac{(-1)^{j-l}}{2\pi\Lambda\Delta}\displaystyle\int_0^\Lambda
 \frac{\cos(k_jt)-\cos(k_lt)}{\sinh(t/2)}dt$ for $j\neq l$, which is \eqref{eq:qoff}.}
\lproof
\lstep{2}{1}{$e^{-ik_jt}(1-e^{i\Delta t})=e^{-ik_jt}-e^{-ik_lt}$.}
\lby{$\Delta=k_j-k_l$}
\lstep{2}{2}{$\operatorname{sgn}(t)\bigl(\cos k_jt-\cos k_lt\bigr)/\sinh(t/2)$ is even and
 $\operatorname{sgn}(t)\bigl(\sin k_jt-\sin k_lt\bigr)/\sinh(t/2)$ is odd.}
\lby{parity}
\lstep{2}{3}{Hence the $t$-integral equals $\frac{(-1)^{j-l}}{i\Delta}\cdot
 2\int_0^\Lambda(\cos k_jt-\cos k_lt)/\sinh(t/2)dt$ and no PV is needed
 (the integrand is $O(t)$ at $0$).}
\lby{$\langle1\rangle4$, $\langle2\rangle1$, $\langle2\rangle2$}
\lstep{2}{4}{$-\frac i{4\pi\Lambda}\cdot\frac{2}{i\Delta}=-\frac1{2\pi\Lambda\Delta}$.}
\lby{arithmetic}
\lqed{2}{5}\lby{$\langle2\rangle3$, $\langle2\rangle4$}

\lstep{1}{7}{$\lim_{t\to0}(\Lambda-t)\sin(k_jt)/\sinh(t/2)=\Lambda\cdot k_jt/(t/2)=2\Lambda k_j$.}
\lby{$\sin u\sim u$, $\sinh u\sim u$}

\lstep{1}{8}{In the uncentred basis the substitution of $\langle1\rangle1$ produces the extra
 factor $e^{-i(k_j-k_l)\xi_0}$, because the defect-free kernel is translation invariant but the
 \emph{basis functions} are not translated with the box.}
\lby{$\langle1\rangle1$ with $e_j$ in place of $\check e_j$}

\lqed{1}{9}\lby{$\langle1\rangle5$--$\langle1\rangle8$}

def Q_box(kappa, Lam, nt=400001):
    k = kappa/(2*np.pi); N = len(k)
    t = np.linspace(0, Lam, nt); w = np.full(nt, Lam/(nt-1)); w[0]=w[-1]=Lam/(2*(nt-1))
    with np.errstate(divide='ignore', invalid='ignore'):
        inv_sh = np.where(t>0, 1/np.sinh(t/2), 0.0)
    diag = np.array([0.5 - (1/(2*np.pi*Lam))*(np.sum(w[1:]*(Lam-t[1:])*np.sin(kj*t[1:])*inv_sh[1:]) + w[0]*2*Lam*kj) for kj in k])   # t=0 limit = 2 Lam k

    with np.errstate(divide='ignore', invalid='ignore'):
        Q = np.where(dk!=0, -sgn/(2*np.pi*Lam*dk)*(Ct[:,None]-Ct[None,:]), 0.0)
    Q[np.arange(N), np.arange(N)] = diag
    return Q

\begin{verdictok}
Equations \eqref{eq:qdiag}--\eqref{eq:qoff} (= eq.~(3.5) of the note, ``\file{eq:Qbox}'') are
\textbf{correct}, including the $t\to0$ endpoint value $2\Lambda k_j$, the sign $(-1)^{j-l}$, the
factor $1/(k_j-k_l)$ and the use of the regularised differences $C_j-C_l$, and
\file{compression_box.Q_box} implements them faithfully.  Trapezoid accuracy: the entries change by
$4.4\times10^{-11}$ between $n_t=4\times10^5$ and $1.6\times10^6$ and by $7.0\times10^{-10}$ between
$10^5$ and $4\times10^5$, i.e.\ clean $O(h^2)$ convergence with an $O(5\times10^{-11})$ residual
error at the default $n_t$.
\end{verdictok}

\begin{verdictgap}
\textbf{Basis inconsistency (\S3.1, displayed definition of $\mathfrak h_N$, line 329 of
\file{fidelity_recovered_quasifree.tex}).}  The note defines
``$e_j=\Lambda^{-1/2}e^{ik_j\xi}\mathbf 1_{[\xi_0-\Lambda/2,\xi_0+\Lambda/2]}$'' and then states
\eqref{eq:qdiag}--\eqref{eq:qoff} for it.  In that basis the correct off-diagonal carries the extra
factor $e^{-i(k_j-k_l)\xi_0}$ (Lemma~\ref{lem:qbox}, $\langle1\rangle8$), which for $a=1,L=2$ is not
$1$.  Both the note and \file{Q_box} give the \emph{centred}-basis matrix.  Taken alone this is only
a notational slip; combined with \S\ref{sec:defect} it is not.  Fix: replace $e^{ik_j\xi}$ by
$e^{ik_j(\xi-\xi_0)}$ in the definition of $\mathfrak h_N$.
\end{verdictgap}

\begin{verdictbreak}

\section{The defect matrix}\label{sec:defect}

\begin{lemma}[Box matrix elements of $\delta_s$]\label{lem:defect}
With $\widetilde D_s(\xi,\xi')=\sqrt{\beta(x)\beta(y)}\,\delta_s(x,y)$ and $\delta_s$ as in
eq.~(1.2) of the note, in the centred basis
\begin{equation}\label{eq:Dbox}
 \check D_{jl}=\frac i{2\pi\Lambda}\int_0^{\Lambda/2}\!\!dp\int_0^{\Lambda/2}\!\!dp'\,
   e^{ik_jp}\sqrt{\beta(x_p)}\;R_s(u_p,v_{p'})\;\sqrt{\beta(y_{p'})}\,e^{ik_lp'}\;+\;\mathrm{h.c.},
\end{equation}
where $x_p=x(\xi_0-p)$, $y_{p'}=x(\xi_0+p')$, $u_p=-x_p>0$, $v_{p'}=y_{p'}>0$.  Moreover
$\widetilde D_s$ depends on $(a,L,s)$ only through $\zeta$ and $\eta=\xi-\xi_0$; at first order in $s$,
\begin{equation}\label{eq:D1kernel}
 \widetilde D^{(1)}(\eta,\eta')=\frac i{2\pi}\,\zeta\;
   \Bigl[-\frac{\sinh\frac\eta2\,\sinh\frac{\eta'}2}{\sinh^2\frac{\eta-\eta'}2}\Bigr].
\end{equation}
\end{lemma}

\lstep{1}{1}{$x+a=\frac{(a+L)e^\xi}{1+e^\xi}$, $L-x=\frac{a+L}{1+e^\xi}$, hence
 $\beta=\frac{(a+L)e^\xi}{(1+e^\xi)^2}$ and $\sqrt\beta=\frac{\sqrt{a+L}\,e^{\xi/2}}{1+e^\xi}$.}
\lby{$x=(Le^\xi-a)/(1+e^\xi)$ from \eqref{eq:conv}}
\lstep{1}{2}{$u/\sqrt\beta=-2L e^{\xi_0/2}\sinh\frac{\xi-\xi_0}2/\sqrt{a+L}$ and
 $v/\sqrt{\beta'}=+2Le^{\xi_0/2}\sinh\frac{\xi'-\xi_0}2/\sqrt{a+L}$, hence
 $uv/\sqrt{\beta\beta'}=-4\beta_0\sinh\frac\eta2\sinh\frac{\eta'}2$, $\beta_0=aL/(a+L)$.}
\lby{$\langle1\rangle1$, $a=Le^{\xi_0}$, and $e^{\xi_0}\cdot L^2/(a+L)=aL/(a+L)$}
\lstep{1}{3}{$u+v=y-x=2\sqrt{\beta\beta'}\sinh\frac{\xi'-\xi}2$.}
\lby{eq.~(3.4) of the note, ``$(x-y)/\sqrt{\beta(x)\beta(y)}=2\sinh\frac{\xi-\xi'}2$'', verified from
 $\langle1\rangle1$}
\lstep{1}{4}{Therefore $\sqrt{\beta\beta'}\,\frac{uv}{(u+v)^2}
 =\frac{uv/\sqrt{\beta\beta'}}{4\sinh^2\frac{\eta-\eta'}2}
 =-\beta_0\frac{\sinh\frac\eta2\sinh\frac{\eta'}2}{\sinh^2\frac{\eta-\eta'}2}$, and with
 $R_s=\frac sL\frac{uv}{(u+v)^2}+O(s^2)$ and $s\beta_0/L=\zeta$ this is \eqref{eq:D1kernel}.}
\lby{$\langle1\rangle2$, $\langle1\rangle3$, and $R_s(u,v)=\frac{suv}{(u+v)(L(u+v)+suv)}$}
\lstep{1}{5}{\eqref{eq:Dbox} is the definition of the matrix element of the block-off-diagonal
 operator $\delta_s=E_A\delta_sE_D+E_D\delta_sE_A$ in the centred basis, after the substitution
 $\xi=\xi_0-p$ on $A$ and $\xi'=\xi_0+p'$ on $D$ (note $d\xi=-dp$ but the orientation of the
 interval is reversed as well, so the sign cancels), and the box restricts $p,p'\le\Lambda/2$.}
\lby{$A=\{\xi<\xi_0\}$, $D=\{\xi>\xi_0\}$ in the modular coordinate; $M+M^\dagger$ is the matrix of
 a block-off-diagonal Hermitian operator whose $A\times D$ block has matrix $M$}
\lqed{1}{6}\lby{$\langle1\rangle4$, $\langle1\rangle5$}

\subsection{Is the sub-compression a compression?}

\begin{proposition}\label{prop:subcomp}
Let $\Pi:\mathbb C^N\to L^2(I)$ be the isometry onto $\mathfrak h_N$, $Q_N=\Pi^*Q\Pi$,
$\Qt_N=\Pi^*\Qt_s\Pi$, and let $P\in\mathbb C^{N\times n}$ have as columns an orthonormal set of
eigenvectors of $Q_N$.  Then $P^*Q_NP$ and $P^*\Qt_NP$ are the compressions of $Q$ and $\Qt_s$ onto
the $n$-dimensional subspace $\Pi P\,\mathbb C^n\subset L^2(I)$; in particular
$-\log\Fid_{\rm sub}\le-\log\Fid$ and $S_{\rm sub}\le S$, and $P^*Q_NP=\operatorname{diag}(w)$ exactly.
\end{proposition}

\lby{$P^*P=1_n$, $\Pi^*\Pi=1_N$}
\lstep{1}{2}{$(\Pi P)^*Q(\Pi P)=P^*\Pi^*Q\Pi P=P^*Q_NP$, and likewise for $\Qt_s$;
 $P^*\Qt_NP=P^*(Q_N-\Dh)P$ \emph{provided} $\Dh$ is the matrix of $\delta_s$ in the same basis as
 $Q_N$.}
\lby{definition of $Q_N$, $\Qt_N$; see the caveat below}
\lstep{1}{3}{Both are strictly between $0$ and $1$, so Theorem 2.1 applies, and Theorem 3.1's

\lby{Theorem 3.1 of the note (fidelity nonincreasing, relative entropy nondecreasing along

\lstep{1}{4}{$P^*Q_NP=\operatorname{diag}(w)$ because the columns of $P$ are eigenvectors of $Q_N$.}

\subsection{The tail correction $\zeta^2/(15\kc^2)$}

\begin{proposition}[Derivation]\label{prop:tailderiv}
Assume (i) the modes with $|\kappa|>\kc$ are decoupled from the kept modes, (ii) they are empty
(resp.\ full) in the vacuum, $q_\kappa\ll1$ (resp.\ $1-q_\kappa\ll1$), and (iii) they carry the
occupation density $\frac2{15}\zeta^2|\kappa|^{-3}$ of Proposition~4.2 of the note.  Then their
contribution to $-\log\Fid$ is $\zeta^2/(15\kc^2)$.
\end{proposition}
\lstep{1}{1}{For a single decoupled mode with commuting occupations $q$ and $\tilde q=q+d$,
 $\Fid=\sqrt{q\tilde q}+\sqrt{(1-q)(1-\tilde q)}$.}
\lby{Theorem 2.1, commuting case (last sentence of Theorem 2.1)}
\lstep{1}{2}{With $q=0$: $\Fid=\sqrt{1-d}$ and $-\log\Fid=-\tfrac12\log(1-d)=\tfrac d2+O(d^2)$.}

 occupation $\frac2{15}\zeta^2\kappa^{-3}\Delta\kappa$, so the discarded particle modes contribute

smallest $\zeta$ of Table~3 ($\kappa_*(1/120)=20.7>18.42=\kc$), and at $\zeta=1/15$,
$\eps=10^{-8}$ the correction is individually $12\%$ too large from the linearisation and $14\%$ too
small from the asymptotic density.  Consequently the sentence
``\emph{the corrected values of $\Phi(1/15)$ agree to $1.4\times10^{-4}$, while the raw values differ
by $1.2\%$; this validates the correction}'' overstates what has been shown: both raw numbers
($1.198\%$ apart) and both corrected numbers ($1.40\times10^{-4}$ apart) are reproduced exactly, but
the agreement rests on a cancellation of a $+12\%$ and a $-14\%$ error in the correction, and it does
not certify the $\zeta\le1/120$ rows where the correction is $56\%$ too large.  A defensible
statement is $\Phi(1/15)=3.517(5)\times10^{-5}$ rather than $3.517(3)\times10^{-5}$.
\end{verdictgap}

\subsection{The $1/\kc$ extrapolation of the relative entropy}

\begin{lemma}\label{lem:extrap}
If $S(\kc)=S_\infty-C/\kc$ and $x_i=1/\kc^{(i)}$, then
$S_\infty=S(x_2)+\bigl(S(x_2)-S(x_1)\bigr)\dfrac{x_2}{x_1-x_2}$.
\end{lemma}
\lstep{1}{1}{$S$ is affine in $x=1/\kc$ with slope $-C=\frac{S(x_2)-S(x_1)}{x_2-x_1}$.}
\lby{hypothesis}
\lstep{1}{2}{$S_\infty=S(0)=S(x_2)-(-C)\,x_2\cdot(-1)=S(x_2)-\frac{S(x_2)-S(x_1)}{x_2-x_1}x_2
 =S(x_2)+\bigl(S(x_2)-S(x_1)\bigr)\frac{x_2}{x_1-x_2}$.}
\lby{$\langle1\rangle1$ and $x_2-x_1=-(x_1-x_2)$}
\lqed{1}{3}\lby{$\langle1\rangle2$}

\begin{lstlisting}[caption={\file{numerics/make_tables.py}, line 43},captionpos=b,firstnumber=43]
        Sinf = c['Ssub'] + (c['Ssub']-r['Ssub'])*(1/c['kc'])/(1/r['kc']-1/c['kc'])
\end{lstlisting}
with \file{r} the $\kc=18.4$ row (\file{results_A.txt}) and \file{c} the $\kc=25.3$ row
(\file{results_hp.txt}) --- exactly Lemma~\ref{lem:extrap}.

\begin{verdictok}
The extrapolation formula is \textbf{correct} for $S=S_\infty-C/\kc$.  Two independent
consistency checks: (i) reproducing the table, $\zeta=1/120$ gives $S_\infty=5.69246\times10^{-6}$
(table: $5.6925\times10^{-6}$), a $15.65\%$ rise over the raw $\kc=25.3$ value (\S6: ``$16\%$'') and
$1.71\%$ below $s_2\zeta^2$ (\S6: ``within $2\%$''); (ii) the fitted law extrapolated \emph{back} to
$\kc=29.61$ predicts $S=5.034\times10^{-6}$, against the independently computed full $91$-mode
compression $5.0207\times10^{-6}$ of \file{results_final.txt} --- $0.27\%$.  Minor: \file{make_tables}
reads the \emph{rounded} $\kc=18.4$, $25.3$ from the text files instead of $18.4207$, $25.3284$; the
induced error is $3\times10^{-5}$ relative and harmless.
\end{verdictok}

\section{Methodological weaknesses, with severity}\label{sec:weak}

\begin{enumerate}[leftmargin=1.6em,itemsep=3pt]
\item[\textbf{W1}] \textbf{(High)} \emph{Basis inconsistency $Q_N$ vs.\ $\Dh$.}  See
 \S\ref{sec:defect}.  It invalidates the ``rigorous lower bound'' status of every tabulated value
 as the code stands, although the numerical shift is only $4.8\times10^{-5}$ relative at
 $\zeta=1/15$.  It is also geometry-dependent: the shift is controlled by $\xi_0=\log(a/L)$ and by
 how far $Q_N$ is from diagonal, so it cannot be assumed small at other $(a,L)$.  One-line fix.
\item[\textbf{W2}] \textbf{(High)} \emph{The tail correction assumes the $\kappa\to\infty$
 asymptotics at $\kappa=\kc$.}  The asymptotic density $\frac2{15}\zeta^2\kappa^{-3}$ is $14\%$ low
 at $\kc=18.4$ and $6.5\%$ low at $\kc=25.3$; the linearisation $\tfrac12\times$occupation is $12\%$
 (resp.\ $56\%$ at $\zeta=1/120$) high.  The residual bias on $\Phi$ is at the $0.1$--$0.9\%$
 level, i.e.\ $2$--$15$ times the quoted $\pm0.06\%$.  The three smallest-$\zeta$ rows of Table~3 at
 $\eps=10^{-8}$ sit outside the regime $\kc>\kappa_*(\zeta)$ that Corollary 4.3 requires.
\item[\textbf{W3}] \textbf{(High)} \emph{$\eps$-dependence is not an error bar.}  The two $\eps$
 levels differ by $1.2\%$ raw; the corrections restore agreement to $1.4\times10^{-4}$ only because
 two $\sim13\%$ errors in the correction cancel (\S\ref{sec:fid}).  The quoted uncertainty
 $3.517(3)\times10^{-5}$ is a $\Lambda,\kmax$ spread; it does not include the tail-model
 systematic.  A third $\eps$ (say $10^{-14}$, requiring $Q_N$ in high precision throughout) would
 settle this.
\item[\textbf{W4}] \textbf{(Medium)} \emph{Noise floors.}  (i) The trapezoid error of \file{Q_box}
 at $n_t=4\times10^5$ is $\approx5\times10^{-11}$ per entry (confirmed against \file{Q_box_mp}:
 $4.7\times10^{-11}$), which is $500\times$ the smallest \emph{kept} eigenvalue $\eps=10^{-8}$ only
 by a factor $200$ --- the smallest kept modes carry a $\sim0.5\%$ relative error in $w_i$, hence in
 $\log w_i$ and in the Kubo--Mori weights.  (ii) The quadrature floor of $\Dh$ is
 $\approx3\times10^{-11}$ relative (GL12 vs GL16), while the ``line'' evaluation multiplies
 $|\Dh_{jl}|^2$ by weights of order $e^{\kmax}$; the note's own rule $\kmax\lesssim\Lambda/2-5$ is
 respected in \file{line_so.txt}, but the two \file{batch5.log} rows
 (\file{Lam240_k60}: $f_2=0.2995$) show what happens when it is not, and no automated guard
 enforces it.
\item[\textbf{W5}] \textbf{(Medium)} \emph{The box edge.}  The sharp cut at $|\eta|=\Lambda/2$ is a
 genuine compression (so the bound direction is safe) but it costs $7.7\%$ of $\|\delta_s\|_1$ at
 $\Lambda=60$ and $3.9\%$ at $\Lambda=240$, i.e.\ the box error decays only like $1/\Lambda$ in the
 trace norm, much more slowly than the $\pm0.06\%$ spread of Table~1 suggests.  Table~1 varies
 $\Lambda$ only over $60$--$120$ and does not exhibit a $\Lambda$-trend, so the $\Lambda\to\infty$
 limit is inferred, not demonstrated.
\item[\textbf{W6}] \textbf{(Medium)} \emph{Treatment of $\kmax$.}  $f_2$ and $s_2$ come from
 four points with a three-parameter law; the exponent is not resolved (\S\ref{sec:indep}), the $s_2$
 error bar is understated by $\approx2\times$, and the $\Lambda=60$ point of Table~2 is fitted
 together with three $\Lambda=120$ points even though the $\Lambda=60$ defect matrix has a
 $10^{-6}$-level box-truncation error (measured against the closed form) --- $50\times$ that of the
 $\Lambda=120$ points.  Dropping the first row changes the fitted $f_2$ by more than the quoted bar.
\item[\textbf{W7}] \textbf{(Low)} \emph{The M\"obius check is an identity.}  Because
 $\widetilde D_s$ depends on $(a,L,s)$ only through $\zeta$ and $\eta$ (Lemma~\ref{lem:defect}), the
 configuration $(2,1,0.1)$ produces the complex conjugate of the matrix of $(1,2,0.2)$
 ($\max\bigl||\Dh_1|-|\Dh_2|\bigr|=3.2\times10^{-15}$).  The test exercises \file{x_of}/\file{beta}
 but cannot detect an error in anything downstream, and it is not evidence for Lemma 1.1.
\item[\textbf{W8}] \textbf{(Low)} \emph{Reproducibility.}  \file{results_final.txt} cannot be
 regenerated with the committed \file{fidelity_flint.py} (different output format, and its $S$
 column is the abandoned full double-precision relative entropy, \texttt{nan} in $7$ of $15$ rows);
 \file{test_pipeline.py} and \file{test_precision.py} raise \texttt{TypeError}; the ``\file{flint}''
 in \file{fidelity_flint.py} is \texttt{mpmath}.  No script records the git state or the mesh
 parameters actually used for each \file{.npz} beyond $(s,\Lambda,\kmax,n_{\rm gl})$.
\item[\textbf{W9}] \textbf{(Low)} \emph{Relative entropy of the discarded modes.}  The code adds
 $2\times$tail to $S_{\rm sub}$, i.e.\ $1\times$occupation per empty mode --- correct to leading
 order --- but \S6 itself argues that for $S$ the occupation tail is \emph{not} the dominant
 remainder (the Kubo--Mori weight grows like $|\kappa_i-\kappa_k|$).  Adding a subleading correction
 and then extrapolating in $1/\kc$ double-counts a little; the effect is below the $1/\kc$
 extrapolation's own uncertainty, but the two corrections are not independent.
\end{enumerate}
