\begin{lemma}[Example]\label{lem:ex} For $0<q<1$: $\;q(1-q)\le\tfrac14$.\end{lemma}
\lstep{1}{1}{\lsuffices\lassume $0<q<1$ \lprove $\tfrac14-q(1-q)\ge0$.}
\lby{definition of $\le$}
\lstep{1}{2}{$\tfrac14-q(1-q)=(q-\tfrac12)^2$.}
\lproof
\lstep{2}{1}{$(q-\tfrac12)^2=q^2-q+\tfrac14$.}
\lby{expanding the square}
\lstep{2}{2}{$\tfrac14-q(1-q)=\tfrac14-q+q^2$.}
\lby{distributivity}
\lqed{2}{3}
\lby{$\langle2\rangle1$, $\langle2\rangle2$}
\lstep{1}{3}{$(q-\tfrac12)^2\ge0$.}
\lby{squares of reals are nonnegative}
\lqed{1}{4}
\lby{$\langle1\rangle1$--$\langle1\rangle3$}
\begin{verdictok}All steps verified.\end{verdictok}
